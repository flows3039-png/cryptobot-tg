Базовые функции для работы с балансами и транзакциями. Используем aiosqlite.

python
import aiosqlite
import logging
from datetime import datetime

DB_PATH = "crypto_bot.db"
log = logging.getLogger(__name__)

async def init_db():
    async with aiosqlite.connect(DB_PATH) as db:
        # Таблица пользователей и их настроек
        await db.execute("""
            CREATE TABLE IF NOT EXISTS users (
                user_id INTEGER PRIMARY KEY,
                username TEXT,
                timezone TEXT DEFAULT 'Europe/Moscow',
                fiat_currency TEXT DEFAULT 'USD'
            )
        """)
        
        # Таблица балансов (у каждого юзера свой баланс в каждой валюте)
        await db.execute("""
            CREATE TABLE IF NOT EXISTS balances (
                user_id INTEGER,
                currency TEXT,
                amount REAL DEFAULT 0.0,
                PRIMARY KEY (user_id, currency)
            )
        """)
        
        # Таблица истории транзакций
        await db.execute("""
            CREATE TABLE IF NOT EXISTS transactions (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_id INTEGER,
                type TEXT, -- deposit, withdraw, swap, p2p, check
                currency TEXT,
                amount REAL,
                status TEXT, -- pending, success, failed
                details TEXT,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)
        await db.commit()
    log.info("База данных инициализирована.")

async def get_user(user_id: int):
    async with aiosqlite.connect(DB_PATH) as db:
        async with db.execute("SELECT username, timezone, fiat_currency FROM users WHERE user_id = ?", (user_id,)) as c:
            return await c.fetchone()

async def create_user_if_not_exists(user_id: int, username: str = ""):
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute("INSERT OR IGNORE INTO users (user_id, username) VALUES (?, ?)", (user_id, username))
        await db.commit()

async def get_balance(user_id: int, currency: str) -> float:
    async with aiosqlite.connect(DB_PATH) as db:
        async with db.execute("SELECT amount FROM balances WHERE user_id = ? AND currency = ?", (user_id, currency)) as c:
            row = await c.fetchone()
            return row[0] if row else 0.0

async def get_all_balances(user_id: int) -> dict:
    async with aiosqlite.connect(DB_PATH) as db:
        async with db.execute("SELECT currency, amount FROM balances WHERE user_id = ? AND amount > 0", (user_id,)) as c:
            rows = await c.fetchall()
            return {row[0]: row[1] for row in rows}

async def update_balance(user_id: int, currency: str, amount: float):
    """Добавляет или списывает средства (amount может быть отрицательным)."""
    async with aiosqlite.connect(DB_PATH) as db:
        # Проверяем, существует ли запись
        async with db.execute("SELECT 1 FROM balances WHERE user_id = ? AND currency = ?", (user_id, currency)) as c:
            exists = await c.fetchone()
            
        if exists:
            await db.execute("UPDATE balances SET amount = amount + ? WHERE user_id = ? AND currency = ?", (amount, user_id, currency))
        else:
            await db.execute("INSERT INTO balances (user_id, currency, amount) VALUES (?, ?, ?)", (user_id, currency, amount))
        await db.commit()

async def log_transaction(user_id: int, tx_type: str, currency: str, amount: float, status: str, details: str = ""):
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute("""
            INSERT INTO transactions (user_id, type, currency, amount, status, details)
            VALUES (?, ?, ?, ?, ?, ?)
        """, (user_id, tx_type, currency, amount, status, details))
        await db.commit()

from aiogram import F, Router
from aiogram.types import Message, CallbackQuery
from aiogram.fsm.context import FSMContext
from aiogram.fsm.state import State, StatesGroup

from keyboards import get_wallet_inline_kb, get_currency_selection_kb
from database import get_all_balances, get_balance
from config import CURRENCIES

router = Router()

class WalletFSM(StatesGroup):
    waiting_for_deposit_amount = State()
    waiting_for_withdraw_amount = State()
    waiting_for_withdraw_address = State()

@router.message(F.text == "💼 Кошелек")
async def show_wallet(message: Message):
    balances = await get_all_balances(message.from_user.id)
    
    text = "👛 **Ваш кошелек**\n\n"
    
    if not balances:
        text += "У вас пока нет активов. Пополните баланс, чтобы начать!\n"
    else:
        total_usd = 0.0
        for ticker, amount in balances.items():
            emoji = CURRENCIES.get(ticker, {}).get("emoji", "🪙")
            text += f"{emoji} **{ticker}**: {amount:.4f}\n"
            
    text += "\nВыберите действие:"
    await message.answer(text, reply_markup=get_wallet_inline_kb(), parse_mode="Markdown")

@router.callback_query(F.data == "wallet:deposit")
async def cb_deposit(callback: CallbackQuery):
    await callback.message.edit_text(
        "Выберите валюту для **пополнения**:", 
        reply_markup=get_currency_selection_kb("deposit"),
        parse_mode="Markdown"
    )

@router.callback_query(F.data.startswith("wallet:withdraw:"))
async def cb_withdraw_currency(callback: CallbackQuery, state: FSMContext):
    ticker = callback.data.split(":")[2]
    balance = await get_balance(callback.from_user.id, ticker)
    
    if balance <= 0:
        await callback.answer(f"У вас нет средств в {ticker}!", show_alert=True)
        return
        
    await state.update_data(currency=ticker)
    await state.set_state(WalletFSM.waiting_for_withdraw_amount)
    
    await callback.message.edit_text(
        f"📤 **Вывод {ticker}**\n\n"
        f"Доступный баланс: **{balance:.4f} {ticker}**\n\n"
        f"Введите сумму для вывода:",
        parse_mode="Markdown"
    )

@router.message(WalletFSM.waiting_for_withdraw_amount)
async def process_withdraw_amount(message: Message, state: FSMContext):
    try:
        amount = float(message.text.replace(",", "."))
        if amount <= 0:

    except ValueError:

        return

    await state.update_data(amount=amount)
    await state.set_state(WalletFSM.waiting_for_withdraw_address)
    
    await message.answer(
        f"Введите адрес кошелька **{ticker}**, куда вы хотите вывести средства:",
        parse_mode="Markdown"
    )

import asyncio
import logging
from aiogram import Bot, Dispatcher
from aiogram.filters import CommandStart
from aiogram.types import Message
from aiogram.fsm.storage.memory import MemoryStorage

from config import BOT_TOKEN
from database import init_db, create_user_if_not_exists
from keyboards import get_main_reply_kb, get_wallet_inline_kb, get_more_menu_kb
import wallet

# Настройка логов
logging.basicConfig(level=logging.INFO)
log = logging.getLogger(__name__)

# Инициализация бота
bot = Bot(token=BOT_TOKEN)
dp = Dispatcher(storage=MemoryStorage())

# Подключаем роутеры
dp.include_router(wallet.router)

@dp.message(CommandStart())
async def cmd_start(message: Message):
    # Создаем юзера в БД при первом входе
    await create_user_if_not_exists(message.from_user.id, message.from_user.username or "")
    
    text = (
        f"👋 Привет, **{message.from_user.first_name}**!\n\n"
        f"💼 Добро пожаловать в **Crypto Bot**.\n"
        f"Здесь вы можете хранить, отправлять и обменивать криптовалюту.\n\n"
        f"👇 Используйте меню ниже для навигации."
    )
    await message.answer(text, reply_markup=get_main_reply_kb(), parse_mode="Markdown")

@dp.message(lambda message: message.text == "⚙️ Ещё")
async def cmd_more(message: Message):
    await message.answer(
        "⚙️ **Дополнительные функции**",
        reply_markup=get_more_menu_kb(),
        parse_mode="Markdown"
    )

async def main():
    await init_db()
    log.info("Бот запущен!")
    await dp.start_polling(bot)

if __name__ == "__main__":
    asyncio.run(main())

async def create_check(user_id: int, currency: str, amount: float, check_type: str, check_id: str):
    """Создает чек в БД. check_type: 'real' или 'fake'"""
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute("""
            INSERT INTO checks (check_id, creator_id, currency, amount, type, is_activated)
            VALUES (?, ?, ?, ?, ?, 0)
        """, (check_id, user_id, currency, amount, check_type))
        await db.commit()

async def get_check(check_id: str):
    async with aiosqlite.connect(DB_PATH) as db:
        async with db.execute("SELECT * FROM checks WHERE check_id = ?", (check_id,)) as c:
            return await c.fetchone()

async def activate_check(check_id: str, user_id: int):
    """Активирует чек и зачисляет баланс юзеру."""
    async with aiosqlite.connect(DB_PATH) as db:
        # Проверяем, не активирован ли уже
        async with db.execute("SELECT is_activated, currency, amount FROM checks WHERE check_id = ?", (check_id,)) as c:
            row = await c.fetchone()
            if not row or row[0] == 1:
                return False, "Чек уже активирован или не найден."
            
            _, currency, amount = row
            # Обновляем статус чека
            await db.execute("UPDATE checks SET is_activated = 1 WHERE check_id = ?", (check_id,))
            await db.commit()
            return True, (currency, amount)

async def init_checks_table():
    """Создает таблицу чеков, если ее нет."""
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute("""
            CREATE TABLE IF NOT EXISTS checks (
                check_id TEXT PRIMARY KEY,
                creator_id INTEGER,
                currency TEXT,
                amount REAL,
                type TEXT,
                is_activated INTEGER DEFAULT 0,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)
        await db.commit()

import uuid
from aiogram import F, Router
from aiogram.types import Message, CallbackQuery
from aiogram.fsm.context import FSMContext
from aiogram.fsm.state import State, StatesGroup

from database import get_balance, update_balance, create_check, log_transaction, get_check, activate_check
from keyboards import get_currency_selection_kb
from config import CURRENCIES

router = Router()

class CheckFSM(StatesGroup):
    waiting_for_amount = State()
    waiting_for_currency = State()

@router.callback_query(F.data == "more:checks")
async def cb_check_start(callback: CallbackQuery):
    text = (
        "🧾 **Создание чека**\n\n"
        "Чек позволяет мгновенно отправить криптовалюту любому пользователю.\n"
        "Выберите валюту для чека:"
    )
    await callback.message.edit_text(
        text, 
        reply_markup=get_currency_selection_kb("check"), 
        parse_mode="Markdown"
    )

@router.callback_query(F.data.startswith("wallet:check:"))
async def cb_check_currency(callback: CallbackQuery, state: FSMContext):
    ticker = callback.data.split(":")[2]
    balance = await get_balance(callback.from_user.id, ticker)
    
    if balance <= 0:
        await callback.answer(f"У вас нет средств в {ticker}!", show_alert=True)
        return
        
    await state.update_data(currency=ticker)
    await state.set_state(CheckFSM.waiting_for_amount)
    
    await callback.message.edit_text(
        f"🧾 **Создание чека в {ticker}**\n\n"
        f"Ваш баланс: **{balance:.4f} {ticker}**\n\n"
        f"Введите сумму чека:",
        parse_mode="Markdown"
    )

@router.message(CheckFSM.waiting_for_amount)
async def process_check_amount(message: Message, state: FSMContext):
    try:
        amount = float(message.text.replace(",", "."))
        if amount <= 0:
            raise ValueError
    except ValueError:
        await message.answer("❌ Введите корректное число.")
        return

    data = await state.get_data()
    ticker = data['currency']
    balance = await get_balance(message.from_user.id, ticker)

    if amount > balance:
        await message.answer(f"❌ Недостаточно средств. Ваш баланс: {balance:.4f} {ticker}")
        return

    # Списываем с баланса
    await update_balance(message.from_user.id, ticker, -amount)
    
    # Генерируем уникальный ID чека
    check_id = str(uuid.uuid4())[:8]
    await create_check(message.from_user.id, ticker, amount, "real", check_id)
    await log_transaction(message.from_user.id, "check_create", ticker, -amount, "success", f"Check ID: {check_id}")
    
    bot_info = await message.bot.get_me()
    check_link = f"https://t.me/{bot_info.username}?start=check_{check_id}"
    
    await message.answer(
        f"✅ **Чек создан!**\n\n"
        f"Сумма: **{amount} {ticker}**\n"
        f"Ссылка для активации:\n`{check_link}`\n\n"
        f"Отправьте эту ссылку получателю.",
        parse_mode="Markdown"
    )
    await state.clear()

@router.message(F.text.startswith("/start check_"))
async def activate_check_link(message: Message):
    """Обработка перехода по ссылке чека."""
    check_id = message.text.split("check_")[1]
    success, result = await activate_check(check_id, message.from_user.id)
    
    if not success:
        await message.answer(f"❌ {result}")
        return
        
    currency, amount = result
    await update_balance(message.from_user.id, currency, amount)
    await log_transaction(message.from_user.id, "check_activate", currency, amount, "success", f"Check ID: {check_id}")
    
    emoji = CURRENCIES.get(currency, {}).get("emoji", "🪙")
    await message.answer(
        f"🎉 **Вы активировали чек!**\n\n"
        f"Вам зачислено: **{amount} {currency} {emoji}**",
        parse_mode="Markdown"
    )

async def cmd_admin(message: Message):

        return # Для остальных бот молчит
    
    kb = InlineKeyboardMarkup(inline_keyboard=[
        [InlineKeyboardButton(text="💰 Выдать себе баланс", callback_data="admin:add_balance")],

async def cb_admin_add_balance(callback: CallbackQuery, state: FSMContext):

    await callback.message.edit_text(
        "Введите сумму и валюту через пробел (например: `1000 USDT`):",
        parse_mode="Markdown"
    )

@router.message(AdminFSM.waiting_for_add_balance)
async def process_admin_add_balance(message: Message, state: FSMContext):
    try:
        parts = message.text.split()
        amount = float(parts[0])
        currency = parts[1].upper()
        
        if currency not in CURRENCIES:
            raise ValueError("Неизвестная валюта")
            
        await update_balance(message.from_user.id, currency, amount)
        await log_transaction(message.from_user.id, "admin_deposit", currency, amount, "success", "Admin action")
        
        await message.answer(f"✅ Выдано **{amount} {currency}**")
        await state.clear()
    except Exception:
        await message.answer("❌ Ошибка формата. Введите: `1000 USDT`", parse_mode="Markdown")

import aiosqlite
import logging
from datetime import datetime

DB_PATH = "cryptobot.db"
log = logging.getLogger(__name__)

async def init_db():
    async with aiosqlite.connect(DB_PATH) as db:
        # Юзеры
        await db.execute("""
            CREATE TABLE IF NOT EXISTS users (
                user_id INTEGER PRIMARY KEY,
                username TEXT,
                timezone TEXT DEFAULT 'Europe/Moscow',
                fiat_currency TEXT DEFAULT 'USD',
                is_verified INTEGER DEFAULT 0
            )
        """)
        # Балансы
        await db.execute("""
            CREATE TABLE IF NOT EXISTS balances (
                user_id INTEGER, currency TEXT, amount REAL DEFAULT 0.0,
                PRIMARY KEY (user_id, currency)
            )
        """)
        # Транзакции
        await db.execute("""
            CREATE TABLE IF NOT EXISTS transactions (
                id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER, type TEXT,
                currency TEXT, amount REAL, status TEXT, details TEXT,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)
        # Чеки
        await db.execute("""
            CREATE TABLE IF NOT EXISTS checks (
                check_id TEXT PRIMARY KEY, creator_id INTEGER, currency TEXT,
                amount REAL, type TEXT, is_activated INTEGER DEFAULT 0,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)
        # P2P Ордера
        await db.execute("""
            CREATE TABLE IF NOT EXISTS p2p_orders (
                id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER, type TEXT,
                fiat TEXT, crypto TEXT, price REAL, amount REAL, limits_min REAL, limits_max REAL,
                payment_method TEXT, status TEXT DEFAULT 'active',
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)
        # Подписки
        await db.execute("""
            CREATE TABLE IF NOT EXISTS subscriptions (
                id INTEGER PRIMARY KEY AUTOINCREMENT, channel_id TEXT, owner_id INTEGER,
                price REAL, currency TEXT, interval TEXT DEFAULT 'month',
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)
        # Розыгрыши
        await db.execute("""
            CREATE TABLE IF NOT EXISTS giveaways (
                id INTEGER PRIMARY KEY AUTOINCREMENT, channel_id TEXT, owner_id INTEGER,
                amount REAL, currency TEXT, participants TEXT DEFAULT '',
                status TEXT DEFAULT 'active', created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)
        await db.commit()
    log.info("DB initialized")

# --- ФУНКЦИИ ---
async def create_user(uid, uname=""):
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute("INSERT OR IGNORE INTO users (user_id, username) VALUES (?, ?)", (uid, uname))
        await db.commit()

async def get_balance(uid, cur):
    async with aiosqlite.connect(DB_PATH) as db:
        async with db.execute("SELECT amount FROM balances WHERE user_id=? AND currency=?", (uid, cur)) as c:
            r = await c.fetchone(); return r[0] if r else 0.0

async def get_all_balances(uid):
    async with aiosqlite.connect(DB_PATH) as db:
        async with db.execute("SELECT currency, amount FROM balances WHERE user_id=? AND amount>0", (uid,)) as c:
            return {r[0]: r[1] for r in await c.fetchall()}

async def update_balance(uid, cur, amt):
    async with aiosqlite.connect(DB_PATH) as db:
        async with db.execute("SELECT 1 FROM balances WHERE user_id=? AND currency=?", (uid, cur)) as c:
            exists = await c.fetchone()
        if exists: await db.execute("UPDATE balances SET amount=amount+? WHERE user_id=? AND currency=?", (amt, uid, cur))
        else: await db.execute("INSERT INTO balances VALUES (?, ?, ?)", (uid, cur, amt))
        await db.commit()

async def log_tx(uid, t_type, cur, amt, status, details=""):
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute("INSERT INTO transactions (user_id, type, currency, amount, status, details) VALUES (?,?,?,?,?,?)",
                         (uid, t_type, cur, amt, status, details))
        await db.commit()

# Чеки
async def create_check(uid, cur, amt, c_type, cid):
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute("INSERT INTO checks (check_id, creator_id, currency, amount, type) VALUES (?,?,?,?,?)",
                         (cid, uid, cur, amt, c_type))
        await db.commit()

async def activate_check(cid):
    async with aiosqlite.connect(DB_PATH) as db:
        async with db.execute("SELECT is_activated, currency, amount FROM checks WHERE check_id=?", (cid,)) as c:
            r = await c.fetchone()
            if not r or r[0] == 1: return False, "Чек не найден или уже активирован"
            await db.execute("UPDATE checks SET is_activated=1 WHERE check_id=?", (cid,))
            await db.commit()
            return True, (r[1], r[2])

async def create_p2p_order(uid, o_type, fiat, crypto, price, amt, lim_min, lim_max, pay):

        await db.execute("INSERT INTO p2p_orders (user_id, type, fiat, crypto, price, amount, limits_min, limits_max, payment_method) VALUES (?,?,?,?,?,?,?,?,?)",
                         (uid, o_type, fiat, crypto, price, amt, lim_min, lim_max, pay))
        await db.commit()

async def get_p2p_orders(o_type, fiat, crypto):

                             (o_type, fiat, crypto)) as c:
            return await c.fetchall()

from aiogram import Router, F
from aiogram.types import Message, CallbackQuery
from aiogram.fsm.context import FSMContext
from aiogram.fsm.state import State, StatesGroup
from config import CURRENCIES
from database import get_all_balances, get_balance, update_balance, log_tx
from keyboards import wallet_inline_kb, currency_select_kb

router = Router()

class WalletFSM(StatesGroup):
    waiting_amount = State()
    waiting_address = State()

@router.message(F.text == "💼 Кошелек")
async def show_wallet(message: Message):
    balances = await get_all_balances(message.from_user.id)
    text = "👛 **Ваш кошелек**\n\n"
    if not balances:
        text += "У вас пока нет активов."
    else:
        for ticker, amt in balances.items():
            emoji = CURRENCIES.get(ticker, {}).get("emoji", "🪙")
            text += f"{emoji} **{ticker}**: {amt:.4f}\n"
    await message.answer(text, reply_markup=wallet_inline_kb(), parse_mode="Markdown")

@router.callback_query(F.data == "wallet:dep")
async def cb_dep(cb: CallbackQuery):
    await cb.message.edit_text("Выберите валюту для **пополнения**:", reply_markup=currency_select_kb("dep"), parse_mode="Markdown")

@router.callback_query(F.data.startswith("wallet:dep:"))
async def cb_dep_cur(cb: CallbackQuery):
    ticker = cb.data.split(":")[2]
    await cb.message.edit_text(
        f"📥 **Пополнение {ticker}**\n\n"
        f"Адрес для пополнения:\n`T{cb.from_user.id}X{ticker}Address123456789`\n\n"
        f"⚠️ Минимум 1 {ticker}.",
        parse_mode="Markdown"
    )

@router.callback_query(F.data == "wallet:wdr")
async def cb_wdr(cb: CallbackQuery):
    await cb.message.edit_text("Выберите валюту для **вывода**:", reply_markup=currency_select_kb("wdr"), parse_mode="Markdown")

@router.callback_query(F.data.startswith("wallet:wdr:"))
async def cb_wdr_cur(cb: CallbackQuery, state: FSMContext):
    ticker = cb.data.split(":")[2]
    bal = await get_balance(cb.from_user.id, ticker)
    if bal <= 0:
        await cb.answer("Нет средств!", show_alert=True); return
    await state.update_data(cur=ticker)
    await state.set_state(WalletFSM.waiting_amount)
    await cb.message.edit_text(f"📤 **Вывод {ticker}**\n\nДоступно: **{bal:.4f}**\n\nВведите сумму:", parse_mode="Markdown")

@router.message(WalletFSM.waiting_amount)
async def proc_wdr_amt(msg: Message, state: FSMContext):
    try: amt = float(msg.text.replace(",", "."))
    except: await msg.answer("❌ Введите число."); return
    data = await state.get_data(); ticker = data['cur']
    bal = await get_balance(msg.from_user.id, ticker)
    if amt > bal or amt <= 0:
        await msg.answer(f"❌ Недостаточно средств (баланс: {bal:.4f})"); return
    await state.update_data(amt=amt)
    await state.set_state(WalletFSM.waiting_address)
    await msg.answer(f"Введите адрес кошелька **{ticker}**:", parse_mode="Markdown")

@router.message(WalletFSM.waiting_address)
async def proc_wdr_addr(msg: Message, state: FSMContext):
    data = await state.get_data()
    await update_balance(msg.from_user.id, data['cur'], -data['amt'])
    await log_tx(msg.from_user.id, "withdraw", data['cur'], -data['amt'], "success", f"To: {msg.text}")
    await msg.answer(f"✅ **Заявка на вывод создана!**\n\n{data['amt']} {data['cur']} будут отправлены в течение 10 минут.", parse_mode="Markdown")
    await state.clear()

@router.callback_query(F.data == "wallet:hist")
async def cb_hist(cb: CallbackQuery):
    await cb.message.edit_text("🧾 **История транзакций**\n\nПока пусто.", reply_markup=wallet_inline_kb(), parse_mode="Markdown")

@router.callback_query(F.data == "wallet:menu")
async def cb_back(cb: CallbackQuery):
    await cb.message.delete()
    await show_wallet(cb.message)

        await msg.answer(text, reply_markup=kb, parse_mode="HTML")

@dp.callback_query(F.data == "menu:main")
async def cb_main(cb: CallbackQuery):
    text = (

        f"{_e('crypto', '💼')} <b>Crypto Bot</b>\n\n"
        f"Выберите раздел:"
    )
    kb = main_menu_inline()
    
    try:
        # Если сообщение с фото, нужно удалить и отправить новое
        await cb.message.delete()
    except:
        pass
        
    img_path = os.path.join(IMAGES_DIR, "main_banner.jpg")
    if os.path.exists(img_path):
        photo = FSInputFile(img_path)
        await cb.message.answer_photo(photo=photo, caption=text, reply_markup=kb, parse_mode="HTML")
    else:
        await cb.message.answer(text, reply_markup=kb, parse_mode="HTML")
    await cb.answer()

@dp.callback_query(F.data == "menu:wallet")
async def cb_wallet(cb: CallbackQuery):
    text = f"{_e('wallet', '👛')} <b>Ваш кошелек</b>\n\n"
    # Тут логика баланса
    from database import get_all_balances
    from config import CURRENCIES
    balances = await get_all_balances(cb.from_user.id)
    
    if not balances:
        text += "У вас пока нет активов."
    else:
        for ticker, amt in balances.items():
            emoji = CURRENCIES.get(ticker, {}).get("emoji", "🪙")
            text += f"{emoji} <b>{ticker}</b>: {amt:.4f}\n"

    kb = wallet_inline_kb()
    img_path = os.path.join(IMAGES_DIR, "wallet_banner.jpg")
    
    try:
        await cb.message.delete()
    except:
        pass
        
    if os.path.exists(img_path):
        photo = FSInputFile(img_path)
        await cb.message.answer_photo(photo=photo, caption=text, reply_markup=kb, parse_mode="HTML")
    else:
        await cb.message.answer(text, reply_markup=kb, parse_mode="HTML")
    await cb.answer()
